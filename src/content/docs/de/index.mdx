import { Card, CardGrid, Aside } from '@astrojs/starlight/components';

<Aside type="caution" title="Frühe Entwicklung">
Kasmin ist in früher Entwicklung (Versionen 0.x) und noch nicht für den Echtbetrieb bereit. Version 1.0 folgt nach
gründlichen Tests.
</Aside>

<CardGrid stagger>
	<Card title="Familienkonten" icon="seti:folder">
		Erziehungsberechtigte verwalten ihre Kinder und deren Daten selbst. Geschwister gehören zu einer Familie;
		Notfallkontakte werden einmal für die ganze Familie eingetragen.
	</Card>
	<Card title="Anmelden ohne Passwort" icon="approve-check">
		Einmal-Anmeldelinks und Codes per E-Mail. Administrative Rollen nutzen zusätzlich einen zweiten Faktor.
	</Card>
	<Card title="Anmeldungen und Warteliste" icon="list-format">
		Plätze und Altersgrenzen je Rolle, automatische Warteliste mit sichtbarer Position, Reserveplätze für besondere Fälle.
	</Card>
	<Card title="Sicherheitsblatt" icon="warning">
		Frei konfigurierbar, einmal im Jahr mit einem Klick bestätigt statt neu ausgefüllt, verschlüsselt und nur für
		wenige Personen sichtbar.
	</Card>
	<Card title="Automatisierung und API" icon="setting">
		Rechnungen mit Geschwisterrabatt, Aussendungen und Erinnerungen, die von selbst laufen – mit Vorschau und
		Freigabe. Alles auch per REST-API und MCP für KI-Assistenten.
	</Card>
	<Card title="Buchhaltung mit Fonds" icon="document">
		Doppelte Buchhaltung eingebaut: Rechnungen und Zahlungen buchen sich selbst, Kostenstellen je Projekt, Fonds, die
		Ergebnisse über die Jahre aufbewahren, Umlage der Verwaltung, Auslagen mit SEPA-Auszahlung, Berichte.
	</Card>
	<Card title="Läuft überall" icon="laptop">
		PHP und MariaDB/MySQL auf normalem Webhosting (Installation mit einer Datei) oder als Docker-Image. Freie
		Software unter der AGPL-3.0.
	</Card>
</CardGrid>
